document.addEventListener("DOMContentLoaded", function() { // 가장 안정적인 원본 rss2json 호출 방식 const rssUrl = "https://rss.blog.naver.com/juuudy_.xml"; const apiUrl = `https://api.rss2json.com/v1/api.json?rss_url=${encodeURIComponent(rssUrl)}`; // 1차 카테고리 방어용 (67번 방 검증) const roomKeywords = ["의료", "인허가", "식품", "해썹", "제조업", "허가", "등록", "신고"]; // 2차 추출용 정밀 키워드 30선 const expandedKeywords = [ "의료기기", "품목", "수입업", "제조업", "해썹", "haccp", "식품", "건강기능식품", "화장품", "의약외품", "인허가", "비영리", "사단법인", "재단법인", "영업등록", "영업신고", "수입식품", "해외제조업소", "장기요양", "재가복지", "공장", "위생", "표기준", "소독업", "축산물", "의료법인", "환자유치", "시설요건", "변경신고", "승인" ]; fetch(apiUrl) .then(response => response.json()) .then(data => { const container = document.getElementById('service2-posts-container'); if (!container) return; container.innerHTML = ''; if (!data.items || data.items.length === 0) { container.innerHTML = '

최신 사례를 불러올 수 없습니다.

'; return; } const filteredPosts = data.items.filter(post => { // 1차: 카테고리(방) 일치 여부 확인 const catName = post.categories ? post.categories.join(" ").toLowerCase() : ""; const isCorrectRoom = roomKeywords.some(kw => catName.includes(kw.toLowerCase())); if (!isCorrectRoom) return false; // 2차: 30개 키워드 매칭 const targetText = (post.title + " " + catName).toLowerCase(); return expandedKeywords.some(keyword => targetText.includes(keyword.toLowerCase())); }).slice(0, 3); if (filteredPosts.length === 0) { container.innerHTML = '

해당 분야의 최신 수임 사례가 없습니다.

'; return; } filteredPosts.forEach((post) => { let tmp = document.createElement("DIV"); tmp.innerHTML = post.description; let summary = tmp.textContent || tmp.innerText || ""; summary = summary.replace(/\s+/g, ' ').trim(); if (summary.length > 320) { summary = summary.substring(0, 320) + "..."; } let categoryName = "의료·인허가 사례"; let pubDate = post.pubDate ? post.pubDate.split(' ')[0] : ''; const cardHtml = `
${categoryName} ${pubDate}

${post.title}

${summary}

※ 본 내용은 블로그 요약본입니다. 전체 진행 과정 및 상세 검토 사항은 원문에서 확인하실 수 있습니다. 블로그에서 전체 내용 보기
`; container.innerHTML += cardHtml; }); }) .catch(error => { console.error('블로그 데이터를 불러오는 중 오류 발생:', error); const container = document.getElementById('service2-posts-container'); if (container) { container.innerHTML = '

사례를 불러오는 중 오류가 발생했습니다.

'; } }); });
